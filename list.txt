local function buildPropertyMap(apiDump)
	local propertyMap = {}
	local superclassMap = {}
	for _, class in ipairs(apiDump.Classes) do
		superclassMap[class.Name] = class.Superclass
	end
	local cachedProperties = {}
	local function getReadableProperties(className)
		if cachedProperties[className] then
			return cachedProperties[className]
		end
		local properties = {}
		for _, class in ipairs(apiDump.Classes) do
			if class.Name == className then
				for _, member in ipairs(class.Members) do
					if member.MemberType == "Property" and member.Security.Read == "None" then
						properties[member.Name] = true
					end
				end
				break
			end
		end
		local super = superclassMap[className]
		if super then
			local inherited = getReadableProperties(super)
			for prop in pairs(inherited) do
				properties[prop] = true
			end
		end
		cachedProperties[className] = properties
		return properties
	end
	for _, class in ipairs(apiDump.Classes) do
		local props = getReadableProperties(class.Name)
		propertyMap[class.Name] = {}

		for prop in pairs(props) do
			table.insert(propertyMap[class.Name], prop)
		end
	end
	return propertyMap
end
local moduleid = 119017964340608
local propertyMap = buildPropertyMap(apiDump)
local function mimicObject(instance,exclusions,children) -- exclusions must be instances (will be included without mimic)
	local className = instance.ClassName
	if className==("ModuleScript") then
		return moduleid
	end
	local mimic = {}
	local props = propertyMap[className] or {}
	for _, propName in pairs(props) do
		local success, value = pcall(function()
			return instance[propName]
		end)
		if success then
            if typeof(value) == "instance" then
                value = mimicObject(value,exclusions,true)
            end
			mimic[propName] = value
		end
	end
	if children then
		for _, child in pairs(instance:GetChildren()) do
			if not table.find(exclusions,child) and not child:IsA"ModuleScript" then
				mimic[child.Name] = mimicObject(child,exclusions,true)
			elseif child:IsA("ModuleScript") then
				mimic[child.Name] = moduleid
			else
				mimic[child.Name] = child
			end
		end
	end
	return mimic
end
function check(tablee,ignorelist)
	for i,v in pairs(tablee) do
		if typeof(v) == "Instance" then
			for i,v in pairs(v:GetDescendants()) do
				if v:IsA("Script") then
					table.insert(ignorelist, v)
				end
			end
			tablee[i] = mimicObject(v,ignorelist,true)
            else
                tablee[i] = v
		end
	end
	return tablee
end
